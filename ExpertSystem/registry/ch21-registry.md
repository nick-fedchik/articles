$$\mathrm{Reasoning} \xrightarrow{\text{Символьний висновок}} \mathrm{Recommendation} \xrightarrow{\text{Декомпозиція HTN}} \mathrm{CandidatePlan} \xrightarrow{\text{Авторизація}} \mathrm{TypedActionContract}$$
